static Preferences prefs;

static String toHex(const uint8_t* buf, size_t len) {
  static const char* hex = "0123456789abcdef";
  String out;
  out.reserve(len * 2);
  for (size_t i = 0; i < len; i++) {
    out += hex[(buf[i] >> 4) & 0xF];
    out += hex[buf[i] & 0xF];
  }
  return out;
}

static bool fromHex(const String& hexStr, uint8_t* out, size_t outLen) {
  if (hexStr.length() != (int)(outLen * 2)) return false;

  auto nibble = [](char c) -> int {
    if (c >= '0' && c <= '9') return c - '0';
    if (c >= 'a' && c <= 'f') return 10 + (c - 'a');
    if (c >= 'A' && c <= 'F') return 10 + (c - 'A');
    return -1;
  };

  for (size_t i = 0; i < outLen; i++) {
    int hi = nibble(hexStr[2 * i]);
    int lo = nibble(hexStr[2 * i + 1]);
    if (hi < 0 || lo < 0) return false;
    out[i] = (uint8_t)((hi << 4) | lo);
  }
  return true;
}

static void sha256(const uint8_t* data, size_t len, uint8_t out[32]) {
  mbedtls_md_context_t ctx;
  const mbedtls_md_info_t* info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);

  mbedtls_md_init(&ctx);
  mbedtls_md_setup(&ctx, info, 0);
  mbedtls_md_starts(&ctx);
  mbedtls_md_update(&ctx, data, len);
  mbedtls_md_finish(&ctx, out);
  mbedtls_md_free(&ctx);
}

// hash = SHA256( salt || password )  (salt 16 bytes)
static String hashWithSaltHex(const uint8_t salt[16], const String& pass) {
  size_t passLen = pass.length();
  if (passLen > 128) passLen = 128;

  uint8_t buf[16 + 128];
  memcpy(buf, salt, 16);
  memcpy(buf + 16, pass.c_str(), passLen);

  uint8_t digest[32];
  sha256(buf, 16 + passLen, digest);
  return toHex(digest, 32);
}

bool authSetCredentials(const String& user, const String& pass) {
  if (user.length() < 3) return false;
  if (pass.length() < 6) return false;

  uint8_t salt[16];
  for (int i = 0; i < 16; i++) {
    salt[i] = (uint8_t)esp_random();
  }

  String saltHex = toHex(salt, 16);
  String hashHex = hashWithSaltHex(salt, pass);

  prefs.putString("user", user);
  prefs.putString("salt", saltHex);
  prefs.putString("hash", hashHex);
  prefs.putBool("configured", true);
  return true;
}

bool authVerify(const String& user, const String& pass) {
  if (!authIsConfigured()) return false;

  String savedUser = prefs.getString("user", "");
  if (user != savedUser) return false;

  String saltHex   = prefs.getString("salt", "");
  String savedHash = prefs.getString("hash", "");
  if (saltHex.length() == 0 || savedHash.length() == 0) return false;

  uint8_t salt[16];
  if (!fromHex(saltHex, salt, 16)) return false;

  String candidate = hashWithSaltHex(salt, pass);
  return candidate == savedHash;
}

bool authClear() {
  // clear semua key di namespace "auth"
  bool ok = prefs.clear();
  // pastikan state "configured" jadi false
  prefs.putBool("configured", false);
  return ok;
}

// src/DetectionEngine.cpp
#include "DetectionEngine.h"
#include "DetectionRules.h"
#include <string.h>

// =================== Konfigurasi Detection ===================

#define MAX_DETECTION_FLOWS   32
#define MAX_ALERTS            16

// Window global per proto akan diset dari rules (maks window per proto)
static uint32_t TCP_WINDOW_MS = 5000;  // default 5 detik
static uint32_t UDP_WINDOW_MS = 5000;  // default 5 detik

// =================== Kalibrasi Traffic (Low/Med/High) ===================
#define CAL_WINDOW_MS   10000  // 10 detik
#define CAL_LOW_PPS     20
#define CAL_HIGH_PPS    80
#define CAL_MULT_LOW    1
#define CAL_MULT_MED    2
#define CAL_MULT_HIGH   3

// =================== Flow Stats ===================

struct FlowStats {
  bool     inUse;
  uint8_t  srcIp[4];     // src IP terakhir yang terlihat
  uint8_t  dstIp[4];
  uint16_t dstPort;
  uint8_t  proto;        // 6 = TCP, 17 = UDP, dsb

  uint32_t firstSeenMs;
  uint32_t lastSeenMs;

  uint32_t pktCount;     // semua paket dalam window
  uint32_t synCount;     // TCP SYN (SYN=1, ACK=0)
  uint32_t ackCount;     // TCP ACK-only (ACK=1, SYN=0)
  uint32_t synAckCount;  // TCP SYN-ACK (SYN=1, ACK=1)

  bool alerted[ATTACK_TYPE_COUNT];  // status alert per AttackType
};

static FlowStats   g_flows[MAX_DETECTION_FLOWS];
static AttackAlert g_alerts[MAX_ALERTS];
static uint16_t    s_alertCount = 0;

enum TrafficLevel {
  TRAFFIC_LOW = 0,
  TRAFFIC_MED,
  TRAFFIC_HIGH
};

static TrafficLevel g_trafficLevel = TRAFFIC_LOW;
static TrafficLevel g_levelCandidate = TRAFFIC_LOW;
static uint8_t      g_levelCandidateCount = 0;
static uint32_t     g_calWinStartMs = 0;
static uint32_t     g_calPktCount = 0;
static bool         g_calibrating = false;
static uint32_t     g_calibEndMs = 0;

static const char* trafficLevelToStr(TrafficLevel lvl) {
  switch (lvl) {
    case TRAFFIC_LOW:  return "LOW";
    case TRAFFIC_MED:  return "MEDIUM";
    case TRAFFIC_HIGH: return "HIGH";
    default:           return "UNKNOWN";
  }
}

// =================== Helper ===================

static uint8_t l4ProtoToIpProto(L4Proto p) {
  switch (p) {
    case L4_TCP:  return 6;
    case L4_UDP:  return 17;
    default:      return 0;
  }
}

static bool ipEqual(const uint8_t a[4], const uint8_t b[4]) {
  return a[0] == b[0] && a[1] == b[1] && a[2] == b[2] && a[3] == b[3];
}

static uint32_t trafficMultiplier() {
  switch (g_trafficLevel) {
    case TRAFFIC_LOW:  return CAL_MULT_LOW;
    case TRAFFIC_MED:  return CAL_MULT_MED;
    case TRAFFIC_HIGH: return CAL_MULT_HIGH;
    default:           return CAL_MULT_LOW;
  }
}

static void updateTrafficLevel(uint32_t tsMs) {
  if (g_calWinStartMs == 0) {
    g_calWinStartMs = tsMs;
  }

  g_calPktCount++;

  uint32_t winMs = tsMs - g_calWinStartMs;
  if (winMs < CAL_WINDOW_MS) return;

  uint32_t pps = (winMs > 0) ? (g_calPktCount * 1000UL / winMs) : 0;
  TrafficLevel newLevel = TRAFFIC_LOW;
  if (pps < CAL_LOW_PPS) {
    newLevel = TRAFFIC_LOW;
  } else if (pps < CAL_HIGH_PPS) {
    newLevel = TRAFFIC_MED;
  } else {
    newLevel = TRAFFIC_HIGH;
  }

  if (newLevel == g_trafficLevel) {
    g_levelCandidateCount = 0;
  } else {
    if (newLevel == g_levelCandidate) {
      if (g_levelCandidateCount < 255) g_levelCandidateCount++;
      if (g_levelCandidateCount >= 2) {
        g_trafficLevel = newLevel;
        g_levelCandidateCount = 0;
      }
    } else {
      g_levelCandidate = newLevel;
      g_levelCandidateCount = 1;
    }
  }

  g_calWinStartMs = tsMs;
  g_calPktCount = 0;
}

// Flow di-track by_dst (dstIp, dstPort, proto)
// srcIp hanya disimpan sebagai "last seen src" untuk laporan alert.
static FlowStats* findOrAllocFlow(const PacketInfo& info, uint32_t tsMs) {
  uint8_t ipProto = l4ProtoToIpProto(info.l4Proto);
  if (ipProto == 0) return nullptr;

  // Cari flow yang cocok (by dst)
  for (int i = 0; i < MAX_DETECTION_FLOWS; i++) {
    FlowStats &f = g_flows[i];
    if (!f.inUse) continue;
    if (f.proto != ipProto) continue;
    if (!ipEqual(f.dstIp, info.dstIp)) continue;
    if (f.dstPort != info.dstPort) continue;
    return &f;
  }

  // Cari slot kosong
  for (int i = 0; i < MAX_DETECTION_FLOWS; i++) {
    FlowStats &f = g_flows[i];
    if (!f.inUse) {
      memset(&f, 0, sizeof(f));
      f.inUse = true;
      memcpy(f.srcIp, info.srcIp, 4);
      memcpy(f.dstIp, info.dstIp, 4);
      f.dstPort = info.dstPort;
      f.proto   = ipProto;
      f.firstSeenMs = f.lastSeenMs = tsMs;
      return &f;
    }
  }

  // Kalau penuh, overwrite flow paling tua (firstSeen paling kecil)
  int oldestIdx = 0;
  uint32_t oldestTs = g_flows[0].firstSeenMs;
  for (int i = 1; i < MAX_DETECTION_FLOWS; i++) {
    if (g_flows[i].firstSeenMs < oldestTs) {
      oldestTs = g_flows[i].firstSeenMs;
      oldestIdx = i;
    }
  }
  FlowStats &f = g_flows[oldestIdx];
  memset(&f, 0, sizeof(f));
  f.inUse = true;
  memcpy(f.srcIp, info.srcIp, 4);
  memcpy(f.dstIp, info.dstIp, 4);
  f.dstPort = info.dstPort;
  f.proto   = ipProto;
  f.firstSeenMs = f.lastSeenMs = tsMs;
  return &f;
}

static void addAlert(AttackType type, const FlowStats& f, uint32_t tsMs, uint32_t counterValue) {
  if (s_alertCount >= MAX_ALERTS) return;
  AttackAlert &a = g_alerts[s_alertCount++];
  a.type = type;
  a.timestampMs = tsMs;
  memcpy(a.srcIp, f.srcIp, 4);
  memcpy(a.dstIp, f.dstIp, 4);
  a.dstPort = f.dstPort;
  a.packetCount = counterValue;
}

// Ambil counter sesuai tipe counter
static uint32_t getCounterValue(const FlowStats& f, CounterType ctype) {
  switch (ctype) {
    case COUNTER_PKT:      return f.pktCount;
    case COUNTER_SYN:      return f.synCount;
    case COUNTER_ACK_ONLY: return f.ackCount;
    case COUNTER_SYNACK:   return f.synAckCount;
    default:               return 0;
  }
}

static void applyRules(FlowStats* f, L4Proto l4, uint32_t tsMs) {
  if (!f) return;

  for (size_t i = 0; i < DETECTION_RULE_COUNT; i++) {
    const SimpleRule& r = DETECTION_RULES[i];

    // Cek proto
    if (r.proto == RULE_PROTO_TCP && l4 != L4_TCP) continue;
    if (r.proto == RULE_PROTO_UDP && l4 != L4_UDP) continue;

    // Cek port (0 = any)
    if (r.dstPort != 0 && f->dstPort != r.dstPort) continue;

    // Cek window (flow window harus tidak lebih besar dari window rule)
    uint32_t windowDur = f->lastSeenMs - f->firstSeenMs;
    if (windowDur > r.windowMs) {
      // kalau window flow sudah lebih besar dari window rule,
      // rule ini akan dinilai setelah counter di-reset di siklus berikutnya.
      continue;
    }

    // Ambil nilai counter
    uint32_t val = getCounterValue(*f, r.counter);

    // Cek threshold dan status alert
    if (r.attack < 0 || r.attack >= ATTACK_TYPE_COUNT) continue;
    if (f->alerted[r.attack]) continue;

    uint32_t mult = trafficMultiplier();
    uint32_t threshold = (uint32_t)((uint64_t)r.thresholdCount * (uint64_t)mult);
    if (threshold == 0) threshold = r.thresholdCount;

    if (val >= threshold) {
      addAlert(r.attack, *f, tsMs, val);
      f->alerted[r.attack] = true;
    }
  }
}

// =================== Public API ===================

void detectionInit() {
  // Hitung window global sederhana per proto:
  // ambil nilai maksimum windowMs dari rule-rule TCP dan UDP.
  uint32_t tcpMax = 0;
  uint32_t udpMax = 0;
  for (size_t i = 0; i < DETECTION_RULE_COUNT; i++) {
    const SimpleRule& r = DETECTION_RULES[i];
    if (r.proto == RULE_PROTO_TCP && r.windowMs > tcpMax) {
      tcpMax = r.windowMs;
    } else if (r.proto == RULE_PROTO_UDP && r.windowMs > udpMax) {
      udpMax = r.windowMs;
    }
  }
  if (tcpMax > 0) TCP_WINDOW_MS = tcpMax;
  if (udpMax > 0) UDP_WINDOW_MS = udpMax;

  detectionReset();
}

void detectionReset() {
  memset(g_flows, 0, sizeof(g_flows));
  memset(g_alerts, 0, sizeof(g_alerts));
  s_alertCount = 0;
  g_trafficLevel = TRAFFIC_LOW;
  g_levelCandidate = TRAFFIC_LOW;
  g_levelCandidateCount = 0;
  g_calWinStartMs = 0;
  g_calPktCount = 0;
  g_calibrating = false;
  g_calibEndMs = 0;
}

void detectionUpdateAndCheck(const PacketInfo& info, uint32_t tsMs) {
  if (!info.hasIp) return;

  // Hanya peduli TCP & UDP
  if (info.l4Proto != L4_TCP && info.l4Proto != L4_UDP) {
    return;
  }

  if (g_calibrating && tsMs >= g_calibEndMs) {
    g_calibrating = false;
  }

  updateTrafficLevel(tsMs);

  if (g_calibrating) {
    return;
  }

  FlowStats* f = findOrAllocFlow(info, tsMs);
  if (!f) return;

  // Update "last seen src" untuk laporan
  memcpy(f->srcIp, info.srcIp, 4);

  // Reset window jika sudah lewat (per proto)
  uint32_t windowMs = (info.l4Proto == L4_TCP) ? TCP_WINDOW_MS : UDP_WINDOW_MS;
  if (tsMs - f->firstSeenMs > windowMs) {
    f->firstSeenMs = tsMs;
    f->pktCount    = 0;
    f->synCount    = 0;
    f->ackCount    = 0;
    f->synAckCount = 0;
    memset(f->alerted, 0, sizeof(f->alerted));
  }

  f->lastSeenMs = tsMs;
  f->pktCount++;

  if (info.l4Proto == L4_TCP) {
    uint8_t flags = info.tcpFlags;
    bool syn = flags & 0x02;
    bool ack = flags & 0x10;

    if (syn && !ack) {
      f->synCount++;
    } else if (syn && ack) {
      f->synAckCount++;
    } else if (!syn && ack) {
      f->ackCount++;
    }

  } else if (info.l4Proto == L4_UDP) {
    // Tidak ada counter tambahan selain pktCount untuk UDP flood generic
  }

  // Setelah counter di-update, terapkan rules
  applyRules(f, info.l4Proto, tsMs);
}

uint16_t detectionGetAlertCount() {
  return s_alertCount;
}

const AttackAlert* detectionGetAlertAt(uint16_t idx) {
  if (idx >= s_alertCount) return nullptr;
  return &g_alerts[idx];
}

const char* detectionGetTrafficLevelStr() {
  return trafficLevelToStr(g_trafficLevel);
}

void detectionStartCalibration(uint32_t durationMs, uint32_t nowMs) {
  g_trafficLevel = TRAFFIC_LOW;
  g_levelCandidate = TRAFFIC_LOW;
  g_levelCandidateCount = 0;
  g_calWinStartMs = 0;
  g_calPktCount = 0;
  g_calibrating = true;
  g_calibEndMs = nowMs + durationMs;
}

bool detectionIsCalibrating() {
  return g_calibrating;
}

// src/DetectionRules.cpp
#include "DetectionRules.h"

// Di sini kita definisikan rule-rule sederhana
// yang terinspirasi style Suricata (proto, flags, track by_dst, count, seconds).

// Catatan:
// - Semua rule ini menggunakan track by_dst yang diimplementasikan
//   di DetectionEngine dengan cara menggabungkan semua sumber (src IP)
//   ke satu flow per (dstIp, dstPort, proto).
// - Threshold & window bisa kamu tuning sesuai kebutuhan lab.

const SimpleRule DETECTION_RULES[] = {
  // TCP SYN Flood
  // proto: tcp, flags: S, track: by_dst, count >= 40 dalam 5 detik
  {
    ATTACK_TCP_SYN_FLOOD,
    RULE_PROTO_TCP,
    COUNTER_SYN,
    0,          // dstPort 0 = semua port
    5000,       // 5 detik
    100          // >= 40 SYN dalam window
  },

  // TCP ACK Flood
  // proto: tcp, flags: A (ACK-only), track: by_dst, count >= 40 dalam 5 detik
  {
    ATTACK_TCP_ACK_FLOOD,
    RULE_PROTO_TCP,
    COUNTER_ACK_ONLY,
    0,
    5000,
    100
  },

  // TCP SYN-ACK Flood (reflected/backscatter flood)
  // proto: tcp, flags: SA, track: by_dst, count >= 40 dalam 5 detik
  {
    ATTACK_TCP_SYNACK_FLOOD,
    RULE_PROTO_TCP,
    COUNTER_SYNACK,
    0,
    5000,
    100
  },

  // UDP Flood (generic)
  // proto: udp, track: by_dst, count >= 100 paket UDP dalam 5 detik
  {
    ATTACK_UDP_FLOOD,
    RULE_PROTO_UDP,
    COUNTER_PKT,
    0,          // semua port
    5000,       // 5 detik
    10000         // >= 100 paket UDP dalam window
  }
};

const size_t DETECTION_RULE_COUNT =
  sizeof(DETECTION_RULES) / sizeof(DETECTION_RULES[0]);

// src/SnifferCore.cpp
#include "SnifferCore.h"
#include "SnifferConfig.h"
#include "DetectionEngine.h"

#include <string.h>

extern "C" {
  #include "esp_wifi.h"
}

// =================== Internal Config ===================

#define PACKET_BUFFER_SIZE 64
#define PACKET_MAX_LEN     256

// =================== Internal Types ===================

// Raw paket hasil capture (buffer dari callback)
struct RawPacket {
  uint32_t timestampMs;
  uint16_t length;
  wifi_promiscuous_pkt_type_t type;
  uint8_t  data[PACKET_MAX_LEN];
};

// =================== Internal State ===================

// Ring buffer untuk raw packet
static RawPacket g_packetBuffer[PACKET_BUFFER_SIZE];
static volatile uint16_t s_head = 0;
static volatile uint16_t s_tail = 0;
static volatile bool     s_overflow = false;

// Ringkasan paket untuk Web UI
static PacketSummary g_packetSummaries[MAX_PACKET_SUMMARY];
static uint16_t      s_summaryCount = 0;

// Statistik
static uint32_t s_processedCount = 0;
static bool     s_calibrationMode = false;

// =================== Helper Ring Buffer ===================

static bool pushRawPacket(const uint8_t* data, uint16_t length, wifi_promiscuous_pkt_type_t type) {
  if (length == 0) return false;
  if (length > PACKET_MAX_LEN) {
    length = PACKET_MAX_LEN; // truncate
  }

  uint16_t nextHead = (s_head + 1) % PACKET_BUFFER_SIZE;
  if (nextHead == s_tail) {
    // buffer full
    s_overflow = true;
    return false;
  }

  RawPacket &slot = g_packetBuffer[s_head];
  slot.timestampMs = millis();
  slot.length      = length;
  slot.type        = type;
  memcpy(slot.data, data, length);

  s_head = nextHead;
  return true;
}

static bool popRawPacket(RawPacket &out) {
  if (s_tail == s_head) {
    return false; // empty
  }
  out = g_packetBuffer[s_tail];
  s_tail = (s_tail + 1) % PACKET_BUFFER_SIZE;
  return true;
}

// =================== Parser 802.11 -> IPv4 + TCP/UDP/ICMP ===================
//
// Asumsi sederhana:
// - hanya proses frame DATA (WIFI_PKT_DATA)
// - offset IPv4 = 32 byte (24 802.11 hdr + 8 LLC/SNAP)
// - hanya IPv4

static bool parsePacket(const RawPacket &rp, PacketInfo &info) {
  info.hasIp   = false;
  info.l4Proto = L4_UNKNOWN;
  memset(info.srcIp, 0, 4);
  memset(info.dstIp, 0, 4);
  info.srcPort = 0;
  info.dstPort = 0;
  info.tcpFlags = 0;
  info.payload = nullptr;
  info.payloadLen = 0;

  if (rp.type != WIFI_PKT_DATA) {
    return false;
  }

  if (rp.length < 40) {
    return false;
  }

  const uint8_t* frame = rp.data;
  const uint16_t ipOffset = 32;

  if (rp.length < ipOffset + 20) {
    return false;
  }

  const uint8_t* ip = frame + ipOffset;

  uint8_t verIhl = ip[0];
  uint8_t version = verIhl >> 4;
  uint8_t ihl     = verIhl & 0x0F;

  if (version != 4) {
    return false;
  }

  uint16_t ipHeaderLen = ihl * 4;
  if (ipHeaderLen < 20 || rp.length < ipOffset + ipHeaderLen) {
    return false;
  }

  uint8_t proto = ip[9];
  memcpy(info.srcIp, ip + 12, 4);
  memcpy(info.dstIp, ip + 16, 4);
  info.hasIp = true;

  const uint8_t* l4 = ip + ipHeaderLen;

  if (proto == 6) { // TCP
    if (rp.length < ipOffset + ipHeaderLen + 20) {
      return true;
    }
    info.l4Proto = L4_TCP;
    info.srcPort = (l4[0] << 8) | l4[1];
    info.dstPort = (l4[2] << 8) | l4[3];
    info.tcpFlags = l4[13];

    uint8_t dataOffset = (l4[12] >> 4) * 4;
    const uint8_t* payload = l4 + dataOffset;
    uint16_t totalLen = ((uint16_t)ip[2] << 8) | ip[3];
    uint16_t headerTotal = ipHeaderLen + dataOffset;
    if (totalLen > headerTotal) {
      info.payloadLen = totalLen - headerTotal;
    } else {
      info.payloadLen = 0;
    }
    info.payload = payload;

  } else if (proto == 17) { // UDP
    if (rp.length < ipOffset + ipHeaderLen + 8) {
      return true;
    }
    info.l4Proto = L4_UDP;
    info.srcPort = (l4[0] << 8) | l4[1];
    info.dstPort = (l4[2] << 8) | l4[3];
    info.tcpFlags = 0;

    const uint8_t* payload = l4 + 8;
    uint16_t totalLen = ((uint16_t)ip[2] << 8) | ip[3];
    uint16_t headerTotal = ipHeaderLen + 8;
    if (totalLen > headerTotal) {
      info.payloadLen = totalLen - headerTotal;
    } else {
      info.payloadLen = 0;
    }
    info.payload = payload;

  } else if (proto == 1) { // ICMP
    info.l4Proto = L4_ICMP;
    info.srcPort = 0;
    info.dstPort = 0;
    info.tcpFlags = 0;

    const uint8_t* payload = l4;
    uint16_t totalLen = ((uint16_t)ip[2] << 8) | ip[3];
    uint16_t headerTotal = ipHeaderLen;
    if (totalLen > headerTotal) {
      info.payloadLen = totalLen - headerTotal;
    } else {
      info.payloadLen = 0;
    }
    info.payload = payload;

  } else {
    info.l4Proto = L4_UNKNOWN;
  }

  return true;
}

// =================== Promiscuous Callback ===================

static void wifiSnifferCallback(void* buf, wifi_promiscuous_pkt_type_t type) {
  const wifi_promiscuous_pkt_t* ppkt = (wifi_promiscuous_pkt_t*)buf;
  const wifi_pkt_rx_ctrl_t& rxCtrl = ppkt->rx_ctrl;

  uint16_t length = rxCtrl.sig_len;
  const uint8_t* payload = ppkt->payload;

  pushRawPacket(payload, length, type);
}

  detectionInit();

  s_processedCount = 0;
  s_summaryCount   = 0;
  s_head = s_tail  = 0;
  s_overflow       = false;

}

void snifferStart(uint8_t channel) {
  // Set channel
  esp_wifi_set_channel(channel, WIFI_SECOND_CHAN_NONE);

  // Filter jenis paket
  wifi_promiscuous_filter_t filter;
  filter.filter_mask = SNIFFER_FILTER_MASK;
  esp_wifi_set_promiscuous_filter(&filter);

  // Daftarkan callback & aktifkan mode promiscuous
  esp_wifi_set_promiscuous_rx_cb(&wifiSnifferCallback);
  esp_wifi_set_promiscuous(true);
}

void snifferStop() {
  esp_wifi_set_promiscuous(false);
}

void snifferProcessBufferedPackets() {
  RawPacket rp;
  PacketInfo info;

  while (popRawPacket(rp)) {
    if (!parsePacket(rp, info)) {
      continue;
    }

    if (info.hasIp) {
      if (s_calibrationMode) {
        // Saat kalibrasi, hanya update traffic level (tanpa statistik/summary)
        detectionUpdateAndCheck(info, rp.timestampMs);
        continue;
      }

      s_processedCount++;

      // Simpan ringkasan untuk Web UI
      if (s_summaryCount < MAX_PACKET_SUMMARY) {
        PacketSummary &ps = g_packetSummaries[s_summaryCount++];
        ps.timestampMs = rp.timestampMs;
        ps.l4Proto     = info.l4Proto;
        memcpy(ps.srcIp, info.srcIp, 4);
        memcpy(ps.dstIp, info.dstIp, 4);
        ps.srcPort     = info.srcPort;
        ps.dstPort     = info.dstPort;
        ps.payloadLen  = info.payloadLen;
      }

      // Kirim ke detection engine untuk feature extraction + rule-based detector
      detectionUpdateAndCheck(info, rp.timestampMs);
    }
  }
}

uint32_t snifferGetProcessedCount() {
  return s_processedCount;
}

  return s_overflow;
}

  return s_summaryCount;
}

  if (index >= s_summaryCount) return nullptr;
  return &g_packetSummaries[index];
}

  s_calibrationMode = enabled;
}

static const char* l4ProtoToString(L4Proto p) {
  switch (p) {
    case L4_TCP:  return "TCP";
    case L4_UDP:  return "UDP";
    case L4_ICMP: return "ICMP";
    default:      return "OTHER";
  }
}

static String ipToString(const uint8_t ip[4]) {
  return String(ip[0]) + "." + String(ip[1]) + "." +
         String(ip[2]) + "." + String(ip[3]);
}

static const char* attackTypeToString(AttackType t) {
  switch (t) {
    case ATTACK_TCP_SYN_FLOOD:    return "TCP SYN Flood";
    case ATTACK_TCP_ACK_FLOOD:    return "TCP ACK Flood";
    case ATTACK_TCP_SYNACK_FLOOD: return "TCP SYN-ACK Flood";
    case ATTACK_UDP_FLOOD:        return "UDP Flood";
    default:                      return "Unknown";
  }
}

// =================== Flow summary (untuk tampilan & CSV) ===================

struct FlowView {
  bool    inUse;
  L4Proto proto;
  uint8_t srcIp[4];
  uint8_t dstIp[4];
  uint16_t srcPort;
  uint16_t dstPort;
  uint32_t pktCount;
  uint32_t totalPayloadBytes;
  uint32_t firstTs;
  uint32_t lastTs;
};

static bool flowEqual(const FlowView &f, const PacketSummary* ps) {
  if (f.proto != ps->l4Proto) return false;
  if (f.srcPort != ps->srcPort) return false;
  if (f.dstPort != ps->dstPort) return false;
  for (int i = 0; i < 4; i++) {
    if (f.srcIp[i] != ps->srcIp[i]) return false;
    if (f.dstIp[i] != ps->dstIp[i]) return false;
  }
  return true;
}

static void buildFlowSummary(FlowView* flows, uint16_t maxFlows, uint16_t &outCount) {
  outCount = 0;
  uint16_t pktCount = snifferGetSummaryCount();

  for (uint16_t i = 0; i < pktCount; i++) {
    const PacketSummary* ps = snifferGetSummaryAt(i);
    if (!ps) continue;

    if (ps->l4Proto != L4_TCP && ps->l4Proto != L4_UDP) continue;

    int foundIdx = -1;
    for (uint16_t j = 0; j < outCount; j++) {
      if (flows[j].inUse && flowEqual(flows[j], ps)) {
        foundIdx = j;
        break;
      }
    }

    if (foundIdx < 0) {
      if (outCount >= maxFlows) {
        continue;
      }
      FlowView &f = flows[outCount];
      f.inUse  = true;
      f.proto  = ps->l4Proto;
      for (int k = 0; k < 4; k++) {
        f.srcIp[k] = ps->srcIp[k];
        f.dstIp[k] = ps->dstIp[k];
      }
      f.srcPort = ps->srcPort;
      f.dstPort = ps->dstPort;
      f.pktCount = 1;
      f.totalPayloadBytes = ps->payloadLen;
      f.firstTs = ps->timestampMs;
      f.lastTs  = ps->timestampMs;
      outCount++;
    } else {
      FlowView &f = flows[foundIdx];
      f.pktCount++;
      f.totalPayloadBytes += ps->payloadLen;
      if (ps->timestampMs < f.firstTs) f.firstTs = ps->timestampMs;
      if (ps->timestampMs > f.lastTs)  f.lastTs  = ps->timestampMs;
    }
  }
}

// Demo TA: 1 session aktif
static String g_sessionSid = "";

static String genToken() {
  uint32_t a = esp_random();
  uint32_t b = esp_random();
  char buf[32];
  snprintf(buf, sizeof(buf), "%08lx%08lx", (unsigned long)a, (unsigned long)b);
  return String(buf);
}

static void sendRedirect(const String& to) {
  server.sendHeader("Location", to);
  server.send(303, "text/plain", "Redirecting...");
}

static bool hasValidSession() {
  if (g_sessionSid.length() == 0) return false;
  String cookie = server.header("Cookie");
  return cookie.indexOf("SID=" + g_sessionSid) >= 0;
}

static void requireLogin() {
  if (!authIsConfigured()) sendRedirect("/setup");
  else sendRedirect("/login");
}

// =================== LED Pattern ===================

void setLed(bool on) {
  digitalWrite(LED_PIN, on ? HIGH : LOW);
}

void updateLedPattern() {
  static uint32_t lastToggle = 0;
  static bool ledOn = false;
  uint32_t now = millis();

  switch (g_state) {
    case STATE_SCAN_SSID:
      if (now - lastToggle >= 300) {
        ledOn = !ledOn;
        setLed(ledOn);
        lastToggle = now;
      }
      break;

    case STATE_CONNECTING:
      if (now - lastToggle >= 150) {
        ledOn = !ledOn;
        setLed(ledOn);
        lastToggle = now;
      }
      break;

    case STATE_WEB_IDLE:
      setLed(true);
      break;

    case STATE_SNIFFING:
      if (now - lastToggle >= 100) {
        ledOn = !ledOn;
        setLed(ledOn);
        lastToggle = now;
      }
      break;

    case STATE_CALIBRATING:
      if (now - lastToggle >= 200) {
        ledOn = !ledOn;
        setLed(ledOn);
        lastToggle = now;
      }
      break;

    case STATE_SNIFF_DONE:
      if (now - lastToggle >= 400) {
        ledOn = !ledOn;
        setLed(ledOn);
        lastToggle = now;
      }
      break;
  }
}

// =================== Scan & Connect ===================

bool scanForTargetSSID() {
  g_state = STATE_SCAN_SSID;

  WiFi.mode(WIFI_STA);
  WiFi.disconnect(true);
  delay(100);

  int n = WiFi.scanNetworks(false, true);
  if (n <= 0) {
    WiFi.scanDelete();
    return false;
  }

  bool found = false;

  for (int i = 0; i < n; i++) {
    String ssid = WiFi.SSID(i);
    if (ssid == TARGET_SSID) {
      g_targetChannel = WiFi.channel(i);
      const uint8_t* bssid = WiFi.BSSID(i);
      if (bssid != nullptr) {
        memcpy(g_targetBSSID, bssid, 6);
        g_hasTargetBSSID = true;
      }
      found = true;
      break;
    }
  }

  WiFi.scanDelete();
  return found;
}

bool connectToTargetSSID() {
  g_state = STATE_CONNECTING;

  WiFi.mode(WIFI_STA);
  WiFi.begin(TARGET_SSID);

  uint32_t start = millis();
  const uint32_t timeoutMs = 15000;

  while (WiFi.status() != WL_CONNECTED && millis() - start < timeoutMs) {
    updateLedPattern();
    delay(100);
  }

  return (WiFi.status() == WL_CONNECTED);
}

// FlowView lokal untuk CSV (duplikat ringan dari WebUI)
struct CsvFlowView {
  bool    inUse;
  L4Proto proto;
  uint8_t srcIp[4];
  uint8_t dstIp[4];
  uint16_t srcPort;
  uint16_t dstPort;
  uint32_t pktCount;
  uint32_t totalPayloadBytes;
  uint32_t firstTs;
  uint32_t lastTs;
};

static bool csvFlowEqual(const CsvFlowView &f, const PacketSummary* ps) {
  if (f.proto != ps->l4Proto) return false;
  if (f.srcPort != ps->srcPort) return false;
  if (f.dstPort != ps->dstPort) return false;
  for (int i = 0; i < 4; i++) {
    if (f.srcIp[i] != ps->srcIp[i]) return false;
    if (f.dstIp[i] != ps->dstIp[i]) return false;
  }
  return true;
}

static void buildCsvFlowSummary(CsvFlowView* flows, uint16_t maxFlows, uint16_t &outCount) {
  outCount = 0;
  uint16_t pktCount = snifferGetSummaryCount();

  for (uint16_t i = 0; i < pktCount; i++) {
    const PacketSummary* ps = snifferGetSummaryAt(i);
    if (!ps) continue;
    if (ps->l4Proto != L4_TCP && ps->l4Proto != L4_UDP) continue;

    int foundIdx = -1;
    for (uint16_t j = 0; j < outCount; j++) {
      if (flows[j].inUse && csvFlowEqual(flows[j], ps)) {
        foundIdx = j;
        break;
      }
    }

    if (foundIdx < 0) {
      if (outCount >= maxFlows) continue;
      CsvFlowView &f = flows[outCount];
      f.inUse  = true;
      f.proto  = ps->l4Proto;
      for (int k = 0; k < 4; k++) {
        f.srcIp[k] = ps->srcIp[k];
        f.dstIp[k] = ps->dstIp[k];
      }
      f.srcPort = ps->srcPort;
      f.dstPort = ps->dstPort;
      f.pktCount = 1;
      f.totalPayloadBytes = ps->payloadLen;
      f.firstTs = ps->timestampMs;
      f.lastTs  = ps->timestampMs;
      outCount++;
    } else {
      CsvFlowView &f = flows[foundIdx];
      f.pktCount++;
      f.totalPayloadBytes += ps->payloadLen;
      if (ps->timestampMs < f.firstTs) f.firstTs = ps->timestampMs;
      if (ps->timestampMs > f.lastTs)  f.lastTs  = ps->timestampMs;
    }
  }
}

static const char* csvProtoToString(L4Proto p) {
  switch (p) {
    case L4_TCP:  return "TCP";
    case L4_UDP:  return "UDP";
    case L4_ICMP: return "ICMP";
    default:      return "OTHER";
  }
}

static String csvIpToString(const uint8_t ip[4]) {
  return String(ip[0]) + "." + String(ip[1]) + "." +
         String(ip[2]) + "." + String(ip[3]);
}

static const char* csvAttackTypeToString(AttackType t) {
  switch (t) {
    case ATTACK_TCP_SYN_FLOOD:    return "TCP_SYN_FLOOD";
    case ATTACK_TCP_ACK_FLOOD:    return "TCP_ACK_FLOOD";
    case ATTACK_TCP_SYNACK_FLOOD: return "TCP_SYNACK_FLOOD";
    case ATTACK_UDP_FLOOD:        return "UDP_FLOOD";
    default:                      return "UNKNOWN";
  }
}

static String buildRawCsv() {
  String csv;
  csv.reserve(4096);
  csv += "timestamp_ms,proto,src_ip,src_port,dst_ip,dst_port,payload_len\r\n";

  uint16_t count = snifferGetSummaryCount();
  for (uint16_t i = 0; i < count; i++) {
    const PacketSummary* ps = snifferGetSummaryAt(i);
    if (!ps) continue;

    csv += String(ps->timestampMs); csv += ",";
    csv += csvProtoToString(ps->l4Proto); csv += ",";
    csv += csvIpToString(ps->srcIp); csv += ",";
    csv += String(ps->srcPort); csv += ",";
    csv += csvIpToString(ps->dstIp); csv += ",";
    csv += String(ps->dstPort); csv += ",";
    csv += String(ps->payloadLen); csv += "\r\n";
  }
  return csv;
}

static String buildFlowsCsv() {
  String csv;
  csv.reserve(4096);
  csv += "proto,src_ip,src_port,dst_ip,dst_port,packet_count,total_payload_bytes,first_ts_ms,last_ts_ms\r\n";

  const uint16_t MAX_FLOWS_CSV = 32;
  CsvFlowView flows[MAX_FLOWS_CSV];
  for (uint16_t i = 0; i < MAX_FLOWS_CSV; i++) flows[i].inUse = false;
  uint16_t flowCount = 0;
  buildCsvFlowSummary(flows, MAX_FLOWS_CSV, flowCount);

  for (uint16_t i = 0; i < flowCount; i++) {
    CsvFlowView &f = flows[i];
    if (!f.inUse) continue;

    csv += csvProtoToString(f.proto); csv += ",";
    csv += csvIpToString(f.srcIp); csv += ",";
    csv += String(f.srcPort); csv += ",";
    csv += csvIpToString(f.dstIp); csv += ",";
    csv += String(f.dstPort); csv += ",";
    csv += String(f.pktCount); csv += ",";
    csv += String(f.totalPayloadBytes); csv += ",";
    csv += String(f.firstTs); csv += ",";
    csv += String(f.lastTs); csv += "\r\n";
  }
  return csv;
}

static String buildAlertsCsv() {
  String csv;
  csv.reserve(2048);
  csv += "timestamp_ms,type,src_ip,dst_ip,dst_port,packet_count\r\n";

  uint16_t alertCount = detectionGetAlertCount();
  for (uint16_t i = 0; i < alertCount; i++) {
    const AttackAlert* a = detectionGetAlertAt(i);
    if (!a) continue;

    csv += String(a->timestampMs); csv += ",";
    csv += csvAttackTypeToString(a->type); csv += ",";
    csv += csvIpToString(a->srcIp); csv += ",";
    csv += csvIpToString(a->dstIp); csv += ",";
    csv += String(a->dstPort); csv += ",";
    csv += String(a->packetCount); csv += "\r\n";
  }
  return csv;
}

// =================== Web Server Handlers ===================

void handleRoot() {
  if (!hasValidSession()) { requireLogin(); return; }
  bool sniffRunning = (g_state == STATE_SNIFFING);
  bool sniffDone    = (g_state == STATE_SNIFF_DONE);
  bool calibrating  = (g_state == STATE_CALIBRATING);
  server.send(200, "text/html", buildMainPage(sniffRunning, sniffDone, calibrating));
}

void handleDownloadRaw() {
  if (!hasValidSession()) { requireLogin(); return; }
  String csv = buildRawCsv();

  server.send(200, "text/csv", csv);
}

  server.sendHeader("Content-Disposition", "attachment; filename=\"flow_summary.csv\"");
  server.send(200, "text/csv", csv);

  server.send(200, "text/csv", csv);

void handleClearOutputs() {
  if (!hasValidSession()) { requireLogin(); return; }

  if (g_state == STATE_SNIFFING) {
    server.send(409, "text/plain", "Sniffer sedang berjalan, tidak bisa clear outputs.");
    return;
  }

  snifferResetStats();
  g_state = STATE_WEB_IDLE;

  sendRedirect("/");
}

void handleCalibrate() {
  if (!hasValidSession()) { requireLogin(); return; }

  if (g_state == STATE_SNIFFING || g_state == STATE_CALIBRATING) {
    server.send(409, "text/plain", "Sniffer/kalibrasi sedang berjalan.");
    return;
  }

  if (!server.hasArg("duration")) {
    server.send(400, "text/plain", "Missing 'duration' parameter");
    return;
  }

  int durSec = server.arg("duration").toInt();
  if (durSec <= 0) durSec = 10;

  g_calibDurationMs = (uint32_t)durSec * 1000UL;
  g_calibStartMs    = millis();

  detectionStartCalibration(g_calibDurationMs, g_calibStartMs);

  String html = "<html><body>";
  html += "<h3>Kalibrasi dimulai selama " + String(durSec) + " detik.</h3>";
  html += "<p>Kalibrasi berjalan terpisah dari sniffer utama.</p>";
  html += "<p><a href='http://esp32-sniffer.local/'><button>Kembali ke Web UI</button></a></p>";
  html += "</body></html>";

  server.send(200, "text/html", html);

  g_state = STATE_CALIBRATING;

  // stop webserver supaya tidak ganggu sniffing kalibrasi
  server.stop();
}

static void startWebServer() {
  // Ambil header Cookie supaya bisa cek session
  const char* keys[] = {"Cookie"};
  server.collectHeaders(keys, 1);

  // Root
  server.on("/", HTTP_GET, handleRoot);

  // ===== Setup (Register sekali) =====
  server.on("/setup", HTTP_GET, []() {
    if (authIsConfigured()) {
      server.send(403, "text/plain", "Setup disabled (already configured).");
      return;
    }

    String html = pageHeader("ESP32 Setup");
    html += "<div class='card'>";
    html += "<h3>Initial Setup (Create Admin)</h3>";
    html += "<p class='hint'>Buat akun admin pertama. Setelah tersimpan, halaman setup akan terkunci.</p>";
    html += "<form method='POST' action='/setup'>";
    html += "Username:";
    html += "<input name='u' minlength='3' required>";
    html += "Password:";
    html += "<input name='p' type='password' minlength='6' required>";
    html += "<button type='submit'>Create Admin</button>";
    html += "</form></div></body></html>";

    server.send(200, "text/html", html);
  });

  server.on("/setup", HTTP_POST, []() {
    if (authIsConfigured()) {
      server.send(403, "text/plain", "Setup disabled (already configured).");
      return;
    }

    String u = server.arg("u");
    String p = server.arg("p");

    if (!authSetCredentials(u, p)) {
      server.send(400, "text/plain", "Invalid credentials (min user 3 chars, pass 6 chars).");
      return;
    }

    sendRedirect("/login");
  });

  // ===== Login =====
  server.on("/login", HTTP_GET, []() {
    if (!authIsConfigured()) {
      sendRedirect("/setup");
      return;
    }

    String html = pageHeader("ESP32 Login");
    html += "<div class='card'>";
    html += "<h3>Login</h3>";
    html += "<form method='POST' action='/login'>";
    html += "Username:";
    html += "<input name='u' required>";
    html += "Password:";
    html += "<input name='p' type='password' required>";
    html += "<button type='submit'>Login</button>";
    html += "</form>";
    html += "<p class='hint'>Session disimpan via cookie. Jika restart ESP32, kamu perlu login lagi.</p>";
    html += "</div></body></html>";

    server.send(200, "text/html", html);
  });

  server.on("/login", HTTP_POST, []() {
    if (!authIsConfigured()) {
      sendRedirect("/setup");
      return;
    }

    String u = server.arg("u");
    String p = server.arg("p");

    if (!authVerify(u, p)) {
      server.send(401, "text/plain", "Login failed.");
      return;
    }

    g_sessionSid = genToken();
    server.sendHeader("Set-Cookie", "SID=" + g_sessionSid + "; Path=/; HttpOnly; SameSite=Lax");
    sendRedirect("/");
  });

  // ===== Logout =====
  server.on("/logout", HTTP_GET, []() {
    g_sessionSid = "";
    server.sendHeader("Set-Cookie", "SID=; Path=/; Max-Age=0; HttpOnly; SameSite=Lax");
    sendRedirect("/login");
  });

  // ===== Reset Auth (hapus admin dari NVS) =====
  server.on("/reset_auth", HTTP_GET, []() {
    if (!hasValidSession()) { requireLogin(); return; }

    // Konfirmasi dulu
    if (!server.hasArg("confirm") || server.arg("confirm") != "1") {
      String html = pageHeader("Reset Admin");
      html += "<div class='card'>";
      html += "<h3>Reset Admin</h3>";
      html += "<p class='hint'>Aksi ini akan menghapus akun admin dari NVS (flash) dan mengaktifkan kembali halaman /setup.</p>";
      html += "<p><b>Konsekuensi:</b> kamu harus membuat admin baru.</p>";
      html += "<a href='/reset_auth?confirm=1'><button>Ya, hapus akun admin</button></a>";
      html += "<a href='/'><button>Batal</button></a>";
      html += "</div></body></html>";
      server.send(200, "text/html", html);
      return;
    }

    // Hapus kredensial
    authClear();

    // Logout juga (hapus cookie)
    g_sessionSid = "";
    server.sendHeader("Set-Cookie", "SID=; Path=/; Max-Age=0; HttpOnly; SameSite=Lax");

    // Balik ke setup
    sendRedirect("/setup");
  });

  // ===== Start sniff =====
  server.on("/start", HTTP_GET, []() {
    if (!hasValidSession()) { requireLogin(); return; }

    if (!server.hasArg("duration")) {
      server.send(400, "text/plain", "Missing 'duration' parameter");
      return;
    }

    int durSec = server.arg("duration").toInt();
    if (durSec <= 0) durSec = 60;

    g_sniffDurationMs = (uint32_t)durSec * 1000UL;
    g_sniffStartMs    = millis();

    snifferResetStats();

    String html = "<html><body>";
    html += "<h3>Sniffer dimulai selama " + String(durSec) + " detik.</h3>";
    html += "<p>Silakan tunggu, lalu kembali ke halaman utama untuk melihat hasil.</p>";
    html += "<p><a href='http://esp32-sniffer.local/'><button>Kembali ke Web UI</button></a></p>";
    html += "</body></html>";

    server.send(200, "text/html", html);

    g_state = STATE_SNIFFING;

    // stop webserver supaya tidak ganggu sniffing
    server.stop();
  });

  // CSV endpoints
  server.on("/download_raw",   HTTP_GET, handleDownloadRaw);
  server.on("/download_flows", HTTP_GET, handleDownloadFlows);
  server.on("/download_alerts",HTTP_GET, handleDownloadAlerts);

  // Clear outputs
  server.on("/clear", HTTP_GET, handleClearOutputs);
  server.on("/calibrate", HTTP_GET, handleCalibrate);

  server.begin();
}

// =================== Setup & Loop ===================

void setup() {
  pinMode(LED_PIN, OUTPUT);
  setLed(false);

  Serial.begin(115200);
  delay(1000);

  // Init auth storage (NVS)
  authInit();

  // Inisialisasi modul sniffer + detection
  snifferInit();

  // Scan SSID target
  bool found = scanForTargetSSID();
  if (!found) g_targetChannel = DEFAULT_SNIFFER_CHANNEL;

  // Connect SSID target
  bool connected = connectToTargetSSID();
  if (!connected) {
    g_state = STATE_SCAN_SSID;
    return;
  }

  // mDNS
  startMdns();

  // Web server
  g_state = STATE_WEB_IDLE;
  startWebServer();
}

void loop() {
  updateLedPattern();

  switch (g_state) {
    case STATE_WEB_IDLE:
      server.handleClient();
      break;

    case STATE_SNIFFING: {
      static bool snifferStarted = false;
      if (!snifferStarted) {
        snifferStarted = true;
        snifferStart(g_targetChannel);
      }

      snifferProcessBufferedPackets();

      uint32_t now = millis();
      if (now - g_sniffStartMs >= g_sniffDurationMs) {
        snifferStop();
        snifferStarted = false;

        snifferProcessBufferedPackets();

        g_state = STATE_SNIFF_DONE;

        // Web server hidup lagi
        startWebServer();
      }
      break;
    }

    case STATE_SNIFF_DONE:
      server.handleClient();
      break;

    case STATE_CALIBRATING: {
      static bool calibStarted = false;
      if (!calibStarted) {
        calibStarted = true;
        snifferSetCalibrationMode(true);
        snifferStart(g_targetChannel);
      }

      snifferProcessBufferedPackets();

      uint32_t now = millis();
      if (now - g_calibStartMs >= g_calibDurationMs) {
        snifferStop();
        snifferSetCalibrationMode(false);
        calibStarted = false;

        snifferProcessBufferedPackets();

        g_state = STATE_WEB_IDLE;

        // Web server hidup lagi
        startWebServer();
      }
      break;
    }

    case STATE_SCAN_SSID:
    case STATE_CONNECTING:
      break;
  }
}

// include/DetectionRules.h
#pragma once
#include <Arduino.h>
#include "DetectionEngine.h"
#include "SnifferCore.h"

// Proto di level rule (disederhanakan)
enum RuleProto {
  RULE_PROTO_TCP = 0,
  RULE_PROTO_UDP
};

// Counter mana yang dipakai rule
enum CounterType {
  COUNTER_PKT = 0,     // total paket dalam window
  COUNTER_SYN,         // jumlah TCP SYN (SYN=1, ACK=0)
  COUNTER_ACK_ONLY,    // jumlah TCP ACK-only (ACK=1, SYN=0)
  COUNTER_SYNACK       // jumlah TCP SYN+ACK (SYN=1, ACK=1)
};

// Simple Suricata-like rule untuk flood detection
struct SimpleRule {
  AttackType  attack;          // jenis serangan (enum AttackType)
  RuleProto   proto;           // TCP / UDP
  CounterType counter;         // counter yang dicek
  uint16_t    dstPort;         // 0 = port apa saja
  uint32_t    windowMs;        // window waktu dalam milidetik
  uint32_t    thresholdCount;  // minimal hitungan counter untuk trigger alert
};

// Koleksi rules yang aktif
extern const SimpleRule DETECTION_RULES[];
extern const size_t     DETECTION_RULE_COUNT;

// include/SnifferCore.h
#pragma once
#include <Arduino.h>

// Layer 4 protocol
enum L4Proto {
  L4_UNKNOWN = 0,
  L4_TCP,
  L4_UDP,
  L4_ICMP
};

// Informasi hasil parsing packet (IP + L4)
struct PacketInfo {
  bool     hasIp;
  L4Proto  l4Proto;
  uint8_t  srcIp[4];
  uint8_t  dstIp[4];
  uint16_t srcPort;
  uint16_t dstPort;
  uint8_t  tcpFlags;    // hanya valid untuk TCP
  const uint8_t* payload;
  uint16_t payloadLen;
};

// Ringkasan paket untuk Web UI
#define MAX_PACKET_SUMMARY 50

struct PacketSummary {
  uint32_t timestampMs;
  L4Proto  l4Proto;
  uint8_t  srcIp[4];
  uint8_t  dstIp[4];
  uint16_t srcPort;
  uint16_t dstPort;
  uint16_t payloadLen;
};

// API Sniffer Core

// Inisialisasi modul sniffer (dipanggil sekali di setup)
void snifferInit();

// Reset statistik & buffer (dipanggil sebelum start sniff baru)
void snifferResetStats();

// Mulai/stop sniffer pada channel tertentu
void snifferStart(uint8_t channel);
void snifferStop();

// Proses semua packet yang ada di ring buffer (dipanggil di loop)
void snifferProcessBufferedPackets();

// Mode kalibrasi: proses paket tanpa menambah statistik/sniffer summary
void snifferSetCalibrationMode(bool enabled);

// Statistik dasar
uint32_t snifferGetProcessedCount();
bool     snifferIsOverflowed();

// Ringkasan paket untuk Web UI
uint16_t             snifferGetSummaryCount();
const PacketSummary* snifferGetSummaryAt(uint16_t index);
